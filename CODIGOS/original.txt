  function makeParticles(count, spread, color, size, opacity) {
    const pos=new Float32Array(count*3), base=new Float32Array(count*3), off=new Float32Array(count*3);
    for(let i=0;i<count;i++){
      const x=(Math.random()-.5)*spread[0], y=(Math.random()-.5)*spread[1], z=(Math.random()-.5)*spread[2];
      pos[i*3]=x; pos[i*3+1]=y; pos[i*3+2]=z;
      base[i*3]=x; base[i*3+1]=y; base[i*3+2]=z;
      off[i*3]=Math.random()*Math.PI*2; off[i*3+1]=Math.random()*Math.PI*2; off[i*3+2]=Math.random()*Math.PI*2;
    }
    const geo=new THREE.BufferGeometry();
    geo.setAttribute('position',new THREE.BufferAttribute(pos,3));
    const pts=new THREE.Points(geo,new THREE.PointsMaterial({color,size,transparent:true,opacity}));
    scene.add(pts);
    return {pts,geo,pos,base,off,count};
  }

  const p1=makeParticles(350,[22,16,12],0x4fa3e8,0.04,0.5);
  const p2=makeParticles(80,[18,12,10],0x88ccff,0.07,0.35);
  const p3=makeParticles(30,[14,10,8],0xffffff,0.1,0.18);

  const r1=new THREE.Mesh(new THREE.TorusGeometry(3.2,0.02,8,80),new THREE.MeshBasicMaterial({color:0x4fa3e8,transparent:true,opacity:0.18}));
  r1.rotation.x=Math.PI*.5; r1.position.y=-1.8; scene.add(r1);
  const r2=new THREE.Mesh(new THREE.TorusGeometry(2.3,0.014,8,80),new THREE.MeshBasicMaterial({color:0x1a6ab5,transparent:true,opacity:0.12}));
  r2.rotation.x=Math.PI*.5; r2.position.y=-1.8; scene.add(r2);
  const r3=new THREE.Mesh(new THREE.TorusGeometry(1.5,0.01,8,80),new THREE.MeshBasicMaterial({color:0x88ccff,transparent:true,opacity:0.09}));
  r3.rotation.x=Math.PI*.5; r3.position.y=-1.8; scene.add(r3);

  /* ── PISO ROCOSO: piedras de distintos tamaños sobre el área de los anillos ── */
  const floorGroup = new THREE.Group();
  floorGroup.position.y = -1.8;

  const rockColors = [0x5a5450, 0x4a463f, 0x6b6259, 0x3c3936, 0x756a5c, 0x474138];

  function makeRock(size, color) {
    // Más subdivisiones = silueta redondeada, no un puñado de triángulos grandes
    const geo = new THREE.IcosahedronGeometry(size, 2);
    const pos = geo.attributes.position;
    // Variación aleatoria por roca, pero el "ruido" en sí es una función continua
    // de la dirección del vértice (no un valor random independiente por vértice),
    // así las superficies vecinas se deforman juntas en vez de crear picos sueltos.
    const a1 = Math.random() * 10, a2 = Math.random() * 10, a3 = Math.random() * 10;
    const f1 = 1.6 + Math.random() * 1.2, f2 = 1.6 + Math.random() * 1.2;
    for (let i = 0; i < pos.count; i++) {
      const nx = pos.getX(i), ny = pos.getY(i), nz = pos.getZ(i);
      const len = Math.sqrt(nx*nx + ny*ny + nz*nz) || 1;
      const dx = nx/len, dy = ny/len, dz = nz/len;
      const wobble = 1
        + 0.11 * Math.sin(dx*f1 + dy*f2 + a1)
        + 0.08 * Math.cos(dz*f2 - dx*f1 + a2)
        + 0.05 * Math.sin(dy*3.2 + dz*2.1 + a3);
      pos.setXYZ(i, nx*wobble, ny*wobble*0.78, nz*wobble);
    }
    geo.computeVertexNormals();
    const mat = new THREE.MeshStandardMaterial({
      color, roughness: 0.92, metalness: 0.04, flatShading: false
    });
    const rock = new THREE.Mesh(geo, mat);
    // Pequeña deformación extra de silueta (alargada/achatada) para variar la forma entre rocas
    rock.scale.set(0.85 + Math.random()*0.4, 0.7 + Math.random()*0.5, 0.85 + Math.random()*0.4);
    rock.castShadow = true;
    rock.receiveShadow = true;
    return rock;
  }

  // Piedras grandes y medianas dispersas sobre el anillo principal
  const ROCK_COUNT = 70;
  for (let i = 0; i < ROCK_COUNT; i++) {
    const angle = Math.random() * Math.PI * 2;
    const radius = 0.9 + Math.random() * 3.0; // entre el centro y más allá del anillo exterior
    const size = 0.045 + Math.pow(Math.random(), 2.2) * 0.32; // más piedras chicas que grandes
    const color = rockColors[Math.floor(Math.random() * rockColors.length)];
    const rock = makeRock(size, color);
    rock.position.set(
      Math.cos(angle) * radius,
      -size * 0.4 + Math.random() * 0.03,
      Math.sin(angle) * radius
    );
    rock.rotation.set(Math.random() * Math.PI, Math.random() * Math.PI, Math.random() * Math.PI);
    floorGroup.add(rock);
  }

  // Grava/piedrecillas pequeñas para dar densidad de terreno
  const GRAVEL_COUNT = 90;
  for (let i = 0; i < GRAVEL_COUNT; i++) {
    const angle = Math.random() * Math.PI * 2;
    const radius = 0.6 + Math.random() * 3.3;
    const size = 0.015 + Math.random() * 0.04;
    const color = rockColors[Math.floor(Math.random() * rockColors.length)];
    const rock = makeRock(size, color);
    rock.position.set(Math.cos(angle) * radius, -size * 0.3, Math.sin(angle) * radius);
    rock.rotation.set(Math.random() * Math.PI, Math.random() * Math.PI, Math.random() * Math.PI);
    floorGroup.add(rock);
  }

  const mainGroup=new THREE.Group(); scene.add(mainGroup);
  new THREE.GLTFLoader().load('./escudo.glb',gltf=>{
    const m=gltf.scene;
    const box=new THREE.Box3().setFromObject(m);
    const center=box.getCenter(new THREE.Vector3());
    const sz=box.getSize(new THREE.Vector3());
    const scale=2.6/Math.max(sz.x,sz.y,sz.z);
    m.scale.setScalar(scale);
    m.position.sub(center.multiplyScalar(scale));
    m.traverse(n=>{if(n.isMesh){n.castShadow=true;n.receiveShadow=true;}});
    mainGroup.add(m);
  },undefined,()=>{});

  let isDrag=false,px=0,py=0,vx=0,vy=0.003,rotY=0.3,rotX=0.1;
  canvas.addEventListener('mousedown',e=>{isDrag=true;px=e.clientX;py=e.clientY;vx=0;vy=0;});
  window.addEventListener('mouseup',()=>{isDrag=false;});
  window.addEventListener('mousemove',e=>{if(!isDrag)return;vy=(e.clientX-px)*.009;vx=(e.clientY-py)*.009;rotY+=vy;rotX=Math.max(-.7,Math.min(.9,rotX+vx));px=e.clientX;py=e.clientY;});

  function animP(p,t,sx,sy,sz){
    for(let i=0;i<p.count;i++){
      p.pos[i*3]=p.base[i*3]+Math.sin(t*sx+p.off[i*3])*.18;
      p.pos[i*3+1]=p.base[i*3+1]+Math.sin(t*sy+p.off[i*3+1])*.22;
      p.pos[i*3+2]=p.base[i*3+2]+Math.cos(t*sz+p.off[i*3+2])*.15;
    }
    p.geo.attributes.position.needsUpdate=true;
  }

  let t=0;
  (function animate(){
    requestAnimationFrame(animate); t+=0.016;
    if(!isDrag){vy*=.96;if(Math.abs(vy)<.0005)vy=.003;rotY+=vy;}
    mainGroup.rotation.y=rotY; mainGroup.rotation.x=rotX;
    mainGroup.position.y=Math.sin(t*.65)*.06;
    r1.rotation.z+=.003; r2.rotation.z-=.002; r3.rotation.z+=.0015;
    animP(p1,t,.4,.3,.35); animP(p2,t,.25,.2,.3); animP(p3,t,.18,.15,.22);
    renderer.render(scene,camera);
  })();

  window.addEventListener('resize',()=>{
    camera.aspect=W()/H(); camera.updateProjectionMatrix(); renderer.setSize(W(),H());

  /* ══════════════════════════════════════════
     SISTEMA DE VISORES 3D POR PANEL
     Cada panel carga su modelo automáticamente
     desde PANEL_MODELS[panelId], igual que el

  // Mapa de viewers: id → { rend, sc, cam, grp, ... }
  const viewers = {};

  /* Partículas decorativas reutilizables para los visores pequeños de cada panel */
  function makeParticlesIn(targetScene, count, spread, color, size, opacity) {
    const pos=new Float32Array(count*3), base=new Float32Array(count*3), off=new Float32Array(count*3);
    for(let i=0;i<count;i++){
      const x=(Math.random()-.5)*spread[0], y=(Math.random()-.5)*spread[1], z=(Math.random()-.5)*spread[2];
      pos[i*3]=x; pos[i*3+1]=y; pos[i*3+2]=z;
      base[i*3]=x; base[i*3+1]=y; base[i*3+2]=z;
      off[i*3]=Math.random()*Math.PI*2; off[i*3+1]=Math.random()*Math.PI*2; off[i*3+2]=Math.random()*Math.PI*2;
    }
    const geo=new THREE.BufferGeometry();
    geo.setAttribute('position',new THREE.BufferAttribute(pos,3));
    const pts=new THREE.Points(geo,new THREE.PointsMaterial({color,size,transparent:true,opacity,depthWrite:false}));
    targetScene.add(pts);
    return {pts,geo,pos,base,off,count};
  }
  function animSmallParticles(p,t,sx,sy,sz,amp){
    for(let i=0;i<p.count;i++){
      p.pos[i*3]  =p.base[i*3]  +Math.sin(t*sx+p.off[i*3])  *amp;
      p.pos[i*3+1]=p.base[i*3+1]+Math.sin(t*sy+p.off[i*3+1])*amp*1.2;
      p.pos[i*3+2]=p.base[i*3+2]+Math.cos(t*sz+p.off[i*3+2])*amp*0.8;
    }
    p.geo.attributes.position.needsUpdate=true;
  }

  function loadPanelModel(panelId) {
    const v = viewers[panelId];
    if (!v) return;
    const path = PANEL_MODELS[panelId];
    const badge = document.getElementById('badge-' + panelId);
    const ph = document.getElementById('ph-' + panelId);
    if (!path) { if(badge) badge.textContent = 'Sin ruta configurada'; return; }

    new THREE.GLTFLoader().load(path, gltf => {
      const m = gltf.scene;
      const box = new THREE.Box3().setFromObject(m);
      const center = box.getCenter(new THREE.Vector3());
      const size = box.getSize(new THREE.Vector3());
      const maxDim = Math.max(size.x, size.y, size.z) || 1;
      // Escalamos el modelo a un tamaño de referencia fijo (1.4 unidades)
      const scale = 1.4 / maxDim;
      m.scale.setScalar(scale);
      m.position.sub(center.multiplyScalar(scale));
      m.traverse(n => { if(n.isMesh){ n.castShadow=true; n.receiveShadow=true; } });
      v.grp.add(m);

      // Cámara a distancia fija generosa: con fov=45° a z=4.2 el campo de
      // visión cubre ~3.5 unidades de alto, de sobra para un modelo de 1.4.
      v.cam.position.set(0, 0, 4.2);
      v.cam.near = 0.1;
      v.cam.far  = 50;
      v.cam.updateProjectionMatrix();
      v.cam.lookAt(0, 0, 0);

      if(ph) ph.style.display = 'none';
      if(badge) badge.textContent = 'Modelo cargado';
    }, undefined, err => {
      console.error('Error cargando ' + path, err);
      if(badge) badge.textContent = 'Error al cargar';
      if(ph){
        const sub = ph.querySelector('.ph3-sub');
        if(sub) sub.textContent = 'No se encontró: ' + path;
      }
    });
  }

  function createViewer(panelId) {
    if (viewers[panelId]) return;

    const cvs = document.getElementById('canvas-' + panelId);
    const W3 = () => cvs.clientWidth  || cvs.offsetWidth  || 300;
    const H3 = () => cvs.clientHeight || cvs.offsetHeight || 300;

    const rend = new THREE.WebGLRenderer({ canvas: cvs, antialias: true, alpha: true });
    rend.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    rend.outputEncoding    = THREE.sRGBEncoding;
    rend.toneMapping       = THREE.ACESFilmicToneMapping;
    rend.toneMappingExposure = 1.3;
    rend.setClearColor(0x020510, 1);

    const sc = new THREE.Scene();
    const cam = new THREE.PerspectiveCamera(45, 1, 0.01, 100);
    cam.position.set(0, 0, 3);

    // Luces
    sc.add(new THREE.AmbientLight(0x1a2a4a, 1.5));
    const dl = new THREE.DirectionalLight(0xadd8ff, 3.0);
    dl.position.set(3, 5, 4); sc.add(dl);
    const fl = new THREE.DirectionalLight(0x4fa3e8, 1.2);
    fl.position.set(-4, 1, -3); sc.add(fl);
    const rl = new THREE.DirectionalLight(0x88ccff, 0.5);
    rl.position.set(0, -3, -4); sc.add(rl);

    const grp = new THREE.Group();
    sc.add(grp);

    // Partículas decorativas de fondo, propias de este visor (no las de la pantalla principal)
    const vp1 = makeParticlesIn(sc, 70, [3.4, 3.4, 3.4], 0x4fa3e8, 0.022, 0.55);
    const vp2 = makeParticlesIn(sc, 26, [2.6, 2.6, 2.6], 0x88ccff, 0.034, 0.4);
    const vp3 = makeParticlesIn(sc, 12, [2.0, 2.0, 2.0], 0xffffff, 0.045, 0.22);

    // Estado drag (rotación con el mouse / touch)
    let drag=false, mpx=0, mpy=0, mvy=0.005, mvx=0, mrotY=0.3, mrotX=0.1;

    function onMD(e){ drag=true; mpx=e.clientX||e.touches[0].clientX; mpy=e.clientY||e.touches[0].clientY; mvy=0; mvx=0; }
    function onMU(){ drag=false; }
    function onMM(e){
      if(!drag) return;
      const cx=e.clientX||(e.touches&&e.touches[0].clientX)||mpx;
      const cy=e.clientY||(e.touches&&e.touches[0].clientY)||mpy;
      mvy=(cx-mpx)*.009; mvx=(cy-mpy)*.009;
      mrotY+=mvy; mrotX=Math.max(-.8,Math.min(.8,mrotX+mvx));
      mpx=cx; mpy=cy;
    }
    cvs.addEventListener('mousedown',onMD);
    cvs.addEventListener('touchstart',e=>{onMD(e);},{passive:true});
    window.addEventListener('mouseup',onMU);
    window.addEventListener('touchend',onMU);
    window.addEventListener('mousemove',onMM);
    window.addEventListener('touchmove',e=>{onMM(e);},{passive:true});

    let lt = 0;
    function loop(ts){
      requestAnimationFrame(loop);
      lt = ts;

      // Resize dinámico
      const w = W3(), h = H3();
      if(rend.domElement.width !== w * rend.getPixelRatio() || rend.domElement.height !== h * rend.getPixelRatio()){
        rend.setSize(w, h);
        cam.aspect = w / Math.max(h, 1);
        cam.updateProjectionMatrix();
      }

      if(!drag){ mvy*=.96; if(Math.abs(mvy)<.0005)mvy=.005; mrotY+=mvy; }
      grp.rotation.y = mrotY;
      grp.rotation.x = mrotX;
      grp.position.y = Math.sin(ts * .001) * 0.05;

      const vt = ts * 0.001;
      animSmallParticles(vp1, vt, .5, .4, .45, .32);
      animSmallParticles(vp2, vt, .35, .3, .4, .26);
      animSmallParticles(vp3, vt, .25, .22, .3, .2);

      rend.render(sc, cam);
    }
    requestAnimationFrame(loop);

    viewers[panelId] = { rend, sc, cam, grp, vp1, vp2, vp3 };

    // Carga automática del modelo asignado a este panel
    loadPanelModel(panelId);
  }

  /* ══════════════════════════════════════════
     LÓGICA DE PANELES
  ══════════════════════════════════════════ */
  const decoLeft  = document.getElementById('deco-left');
  const decoRight = document.getElementById('deco-right');
  const allFbtns  = document.querySelectorAll('.ficon-btn');

  window.openPanel = id => {
    document.querySelectorAll('.panel').forEach(p => p.classList.remove('open'));
    document.getElementById('panel-' + id).classList.add('open');
    allFbtns.forEach(b => b.classList.remove('active'));
    const btn = document.getElementById('fbtn-' + id);
    if(btn) btn.classList.add('active');
    decoLeft.classList.add('hidden');
    decoRight.classList.add('hidden');

    // Ocultamos por completo el canvas de fondo (escena principal) para que
    // nunca se vea detrás de ningún panel, sea cual sea el motivo visual.
    canvas.style.visibility = 'hidden';

    // Crear el viewer (y cargar su modelo) la primera vez que se abre el panel
    setTimeout(() => createViewer(id), 50);
  };

  window.closePanel = () => {
    document.querySelectorAll('.panel').forEach(p => p.classList.remove('open'));
    allFbtns.forEach(b => b.classList.remove('active'));
    decoLeft.classList.remove('hidden');
    decoRight.classList.remove('hidden');
    canvas.style.visibility = 'visible';
  };

})();
